local ballsFolder = workspace:FindFirstChild("Pitch") and workspace.Pitch:FindFirstChild("Balls")

RunService.Heartbeat:Connect(function()
    if not isEnabled then return end
    
    local character = lplr.Character
    if not character then return end
    
    if not ballsFolder then 
        ballsFolder = workspace:FindFirstChild("Pitch") and workspace.Pitch:FindFirstChild("Balls")
        return 
    end

    local bodyPartsMap = {
        Legs = { character:FindFirstChild("RL"), character:FindFirstChild("LL"), character:FindFirstChild("Right Leg"), character:FindFirstChild("Left Leg") },
        Head = { character:FindFirstChild("Head") },
        Torso = { character:FindFirstChild("Torso"), character:FindFirstChild("UpperTorso"), character:FindFirstChild("LowerTorso") }
    }

    for _, obj in ipairs(ballsFolder:GetChildren()) do
        if obj.Name == "VEF" and obj:IsA("BasePart") then
            for categoryName, config in pairs(settingsData) do
                if config.On then
                    local partsList = bodyPartsMap[categoryName]
                    if partsList then
                        for _, part in ipairs(partsList) do
                            if part then
                                local mag = (part.Position - obj.Position).Magnitude
                                if mag <= config.Radius then
                                    firetouchinterest(part, obj, 0)
                                    firetouchinterest(part, obj, 1)
                                end
                            end
                        end
                    end
                end
            end
        end
    end
end)
